31// Bound both buffered bytes and the number of cache operations per fill. A
32// configured chunk larger than this budget still fills one chunk at a time.
33const FILL_BYTES: u64 = 4 * 1024 * 1024;
34const FILL_CHUNKS: u64 = 16;
35
36#[derive(Clone, Copy, Debug, Hash, PartialEq, Eq)]
37pub(crate) struct Snapshot {
38    pub(crate) generation: u64,
39    pub(crate) epoch: u64,
40}

42#[derive(Clone)]
43pub(crate) struct CachedReader {
44    pub(crate) state: Arc<State>,
45    pub(crate) source: Servicer,
46    pub(crate) context: OperationContext,
47    pub(crate) path: String,
48    pub(crate) args: OpRead,
49    pub(crate) subject: Subject,
50    pub(crate) chunk_size: u64,
51    pub(crate) ttl: Duration,
52}

54#[derive(Clone, Debug)]
55pub(crate) enum FillError {
56    Changed,
57    Unverifiable(Arc<Error>),
58    Origin(Arc<Error>),
59}
60
61impl From<Error> for FillError {
62    fn from(error: Error) -> Self {
63        Self::Origin(Arc::new(error))
64    }
65}
66
67impl FillError {
68    fn into_error(self) -> Error {
69        match self {
70            Self::Changed => Error::new(ErrorKind::Unexpected, "object changed during cache read")
71                .set_temporary(),
72            Self::Unverifiable(error) => Error::new(
73                ErrorKind::Unexpected,
74                "cannot verify object identity during cache read",
75            )
76            .set_temporary()
77            .set_source(error),
78            Self::Origin(error) => Arc::try_unwrap(error).unwrap_or_else(|error| {
79                let shared = Error::new(error.kind(), error.message());
80                let shared = if error.is_temporary() {
81                    shared.set_temporary()
82                } else if error.is_persistent() {
83                    shared.set_persistent()
84                } else {
85                    shared
86                };
87                shared.set_source(error)
88            }),
89        }
90    }
91}

125pub(crate) type DiscoveryResult = std::result::Result<Option<Discovery>, FillError>;
126pub(crate) type FillResult = std::result::Result<Unit, FillError>;
127type Prepared = std::result::Result<Option<PreparedStream>, FillError>;

129impl CachedReader {
130    fn unit_chunks(&self) -> u64 {
131        (FILL_BYTES / self.chunk_size).clamp(1, FILL_CHUNKS)
132    }
133
134    fn unit_end(&self, start: u64, end: u64) -> u64 {
135        let span = self.chunk_size.saturating_mul(self.unit_chunks());
136        let last = ((end - 1) / self.chunk_size).saturating_add(1);
137        start
138            .saturating_add(span)
139            .min(last.saturating_mul(self.chunk_size))
140    }
141
142    fn normalized(&self, range: BytesRange, length: u64) -> Option<Range<u64>> {
143        match range {
144            BytesRange::Range { offset, size } if offset < length && size != Some(0) => {
145                Some(offset..size.map_or(length, |size| offset.saturating_add(size).min(length)))
146            }
147            BytesRange::Suffix { size } if size > 0 && length > 0 => {
148                Some(length.saturating_sub(size)..length)
149            }
150            _ => None,
151        }
152    }
153
154    fn origin(&self) -> Result<Reader> {
155        self.source
156            .read(&self.context, &self.path, self.args.clone())
157    }

159    async fn stat(&self, version: Option<&str>) -> Result<Metadata> {
160        let options = StatOptions {
161            version: version.map(str::to_owned),
162            ..Default::default()
163        };
164        Ok(self
165            .source
166            .stat(&self.context, &self.path, OpStat::from(options))
167            .await?
168            .into_metadata())
169    }

171    async fn prepare(&self, range: BytesRange) -> Prepared {
172        if range.size() == Some(0) {
173            return Ok(None);
174        }
175        let Some(generation) = self.state.generation().await else {
176            return Ok(None);
177        };
178        let snapshot = Snapshot {
179            generation,
180            epoch: self.state.epoch(&self.subject).await,
181        };
182        let cached = self.state.pointer(generation, &self.subject).await;
183        if let Some(pointer) = cached {
184            if pointer.is_fresh(self.ttl, self.args.version()) {
185                if pointer.identity.tier != Tier::Nonce {
186                    return self.prepare_verified(snapshot, pointer, range, None).await;
187                }
188                if let Some(stream) = self.cached_nonce(snapshot, pointer.clone(), range).await {
189                    return Ok(Some(PreparedStream::Cached(stream)));
190                }
191                let length = pointer.metadata.content_length();
192                let Some(bytes) = self.normalized(range, length) else {
193                    return Ok(None);
194                };
195                // Fill missing chunks under the existing fresh Tier 4 pointer
196                // without rotating the nonce or orphaning already-cached chunks.
197                return self
198                    .prepare_nonce_fill(snapshot, Some(pointer), length, bytes)
199                    .await;
200            } else if pointer.identity.tier != Tier::Nonce {
201                let observed_at = Timestamp::now();
202                match self.stat(self.args.version()).await {
203                    Ok(metadata) => {
204                        let current = Pointer::new(metadata, observed_at);
205                        if current.identity.tier != Tier::Nonce {
206                            self.state.publish(snapshot, &self.subject, &current).await;
207                            return self.prepare_verified(snapshot, current, range, None).await;
208                        }
209                    }
210                    Err(error) if error.kind() != ErrorKind::NotFound => return Ok(None),
211                    Err(_) => {
212                        self.state.discard(snapshot, &self.subject, &pointer).await;
213                        return Ok(None);
214                    }
215                }
216            }
217            self.state.discard(snapshot, &self.subject, &pointer).await;
218        }
219
220        let forward = match range {
221            BytesRange::Range { offset, size } => {
222                offset..size.map_or(u64::MAX, |size| offset.saturating_add(size))
223            }
224            BytesRange::Suffix { .. } => {
225                // Resolve total length via `stat` when available to avoid an extra
226                // GET whose body would be discarded before aligned chunk fetching.
227                let observed_at = Timestamp::now();
228                let mut metadata = None;
229                if self.source.capability().stat {
230                    match self.stat(self.args.version()).await {
231                        Ok(meta) if meta.is_file() => metadata = Some(meta),
232                        Ok(_) | Err(_) => return Ok(None),
233                    }
234                } else if self.source.capability().read_with_suffix {
235                    let limit = BytesRange::suffix(range.size().unwrap().min(self.chunk_size));
236                    match self.origin()?.open(limit).await {
237                        Ok((response, _)) => {
238                            metadata = response.into_metadata().filter(Metadata::is_file);
239                        }
240                        Err(error) if error.kind() == ErrorKind::RangeNotSatisfied => {
241                            return Ok(None);
242                        }
243                        Err(error) => return Err(error.into()),
244                    }
245                }
246                let Some(metadata) = metadata else {
247                    return Ok(None);
248                };
249                if Identity::verifiable(&metadata).is_some() {
250                    let pointer = Pointer::new(metadata, observed_at);
251                    self.state.publish(snapshot, &self.subject, &pointer).await;
252                    return self.prepare_verified(snapshot, pointer, range, None).await;
253                }
254                let Some(forward) = self.normalized(range, metadata.content_length()) else {
255                    return Ok(None);
256                };
257                forward
258            }
259        };
260        if forward.is_empty() {
261            return Ok(None);
262        }
263        let start = forward.start / self.chunk_size * self.chunk_size;
264        let aligned = start..self.unit_end(start, forward.end);
265        let keys = (start / self.chunk_size..=(aligned.end - 1) / self.chunk_size)
266            .map(|index| DiscoveryKey {
267                snapshot,
268                subject: self.subject,
269                index,
270                chunk_size: self.chunk_size,
271            })
272            .collect();
273        let reader = self.clone();
274        let Some(result) = self
275            .state
276            .discoveries
277            .run(self.context.executor(), keys, move |count| async move {
278                let end = start.saturating_add(reader.chunk_size.saturating_mul(count as u64));
279                reader.discover(snapshot, start..end.min(aligned.end)).await
280            })
281            .await
282        else {
283            return Ok(None);
284        };
285        let Some(discovery) = result? else {
286            return Ok(None);
287        };
288        if discovery.pointer.identity.tier == Tier::Nonce {
289            let Some(bytes) = self.normalized(range, discovery.pointer.metadata.content_length())
290            else {
291                return Ok(None);
292            };
293            if let Some(unit) = discovery
294                .unit
295                .filter(|unit| unit.start <= bytes.start && unit.end() >= bytes.end)
296            {
297                return Ok(Some(PreparedStream::Cached(CachedStream::new(
298                    self.clone(),
299                    snapshot,
300                    discovery.pointer,
301                    bytes,
302                    vec![unit],
303                ))));
304            }
305            return self
306                .prepare_nonce_fill(
307                    snapshot,
308                    None,
309                    discovery.pointer.metadata.content_length(),
310                    bytes,
311                )
312                .await;
313        }
314        self.prepare_verified(snapshot, discovery.pointer, range, discovery.unit)
315            .await
316    }

318    async fn prepare_nonce_fill(
319        &self,
320        snapshot: Snapshot,
321        existing: Option<Pointer>,
322        expected_length: u64,
323        requested: Range<u64>,
324    ) -> Prepared {
325        let aligned_start = requested.start / self.chunk_size * self.chunk_size;
326        let aligned_end = (((requested.end - 1) / self.chunk_size) + 1)
327            .saturating_mul(self.chunk_size)
328            .min(expected_length);
329        let observed_at = Timestamp::now();
330        let origin = self.origin()?;
331        let (response, inner) = match origin
332            .open(BytesRange::new(
333                aligned_start,
334                Some(aligned_end - aligned_start),
335            ))
336            .await
337        {
338            Ok(opened) => opened,
339            Err(error) => {
340                let error = fill_open_error(error);
341                if matches!(error, FillError::Changed)
342                    && let Some(pointer) = &existing
343                {
344                    self.state.discard(snapshot, &self.subject, pointer).await;
345                }
346                return Err(error);
347            }
348        };
349        let Some(metadata) = response.into_metadata().filter(Metadata::is_file) else {
350            return Ok(None);
351        };
352        if metadata.content_length() != expected_length || Identity::verifiable(&metadata).is_some()
353        {
354            if let Some(pointer) = &existing {
355                self.state.discard(snapshot, &self.subject, pointer).await;
356            }
357            return Err(FillError::Changed);
358        }
359        let publish_pointer = existing.is_none();
360        let pointer = existing.unwrap_or_else(|| Pointer::new(metadata, observed_at));
361        let prefix = data_prefix(
362            snapshot.generation,
363            &self.subject,
364            pointer.identity,
365            self.chunk_size,
366        );
367        Ok(Some(PreparedStream::NonceFill(NonceFillStream {
368            reader: self.clone(),
369            snapshot,
370            pointer,
371            publish_pointer,
372            prefix,
373            inner,
374            aligned_offset: aligned_start,
375            aligned_end,
376            requested,
377            chunk_index: aligned_start / self.chunk_size,
378            chunk_parts: Vec::new(),
379            chunk_len: 0,
380            done: false,
381            terminal: None,
382        })))
383    }

385    async fn discover(&self, snapshot: Snapshot, aligned: Range<u64>) -> DiscoveryResult {
386        let observed_at = Timestamp::now();
387        let origin = self.origin()?;
388        let (response, mut stream) = match origin
389            .open(BytesRange::new(
390                aligned.start,
391                Some(aligned.end - aligned.start),
392            ))
393            .await
394        {
395            Ok(result) => result,
396            Err(error) if error.kind() == ErrorKind::RangeNotSatisfied => return Ok(None),
397            Err(error) => return Err(error.into()),
398        };
399        let response = response.into_metadata().filter(Metadata::is_file);
400        let mut use_body = true;
401        let metadata = if let Some(metadata) = response
402            .as_ref()
403            .filter(|metadata| Identity::verifiable(metadata).is_some())
404        {
405            metadata.clone()
406        } else if self.source.capability().stat {
407            match self.stat(self.args.version()).await {
408                Ok(metadata) if metadata.is_file() => {
409                    if response.as_ref().is_some_and(|response| {
410                        response.content_length() != metadata.content_length()
411                    }) {
412                        return Err(FillError::Changed);
413                    }
414                    use_body = response.is_some() && Identity::verifiable(&metadata).is_none();
415                    // Only the metadata returned with these bytes can describe
416                    // a validator-free snapshot. Stat still confirms length.
417                    if use_body {
418                        response.unwrap()
419                    } else {
420                        metadata
421                    }
422                }
423                Ok(_) => return Ok(None),
424                Err(error) if error.kind() == ErrorKind::NotFound => {
425                    return if response.is_some() {
426                        Err(FillError::Changed)
427                    } else {
428                        Ok(None)
429                    };
430                }
431                Err(error) if error.kind() == ErrorKind::Unsupported => match response {
432                    Some(metadata) => metadata,
433                    None => return Ok(None),
434                },
435                Err(error) => return Err(FillError::Unverifiable(Arc::new(error))),
436            }
437        } else {
438            match response {
439                Some(metadata) => metadata,
440                None => return Ok(None),
441            }
442        };
443        let pointer = Pointer::new(metadata, observed_at);
444        let length = pointer.metadata.content_length();
445        if aligned.start >= length {
446            return Ok(None);
447        }
448        let unit = if use_body {
449            let end = aligned.end.min(length);
450            let data = read_unit(&mut stream, end - aligned.start)
451                .await
452                .map_err(fill_open_error)?;
453            if data.len() as u64 != end - aligned.start {
454                return Err(FillError::Changed);
455            }
456            let unit = Unit {
457                start: aligned.start,
458                data,
459            };
460            self.store_unit(snapshot, &pointer, &unit).await;
461            Some(unit)
462        } else {
463            if pointer.identity.tier == Tier::Nonce {
464                return Ok(None);
465            }
466            None
467        };
468        self.state.publish(snapshot, &self.subject, &pointer).await;
469        Ok(Some(Discovery { pointer, unit }))
470    }

472    async fn prepare_verified(
473        &self,
474        snapshot: Snapshot,
475        pointer: Pointer,
476        range: BytesRange,
477        unit: Option<Unit>,
478    ) -> Prepared {
479        let Some(bytes) = self.normalized(range, pointer.metadata.content_length()) else {
480            return Ok(None);
481        };
482        let unit = unit.filter(|unit| unit.start <= bytes.start && bytes.start < unit.end());
483        let mut stream = CachedStream::new(
484            self.clone(),
485            snapshot,
486            pointer,
487            bytes,
488            unit.into_iter().collect(),
489        );
490        if stream.units.is_empty() {
491            match stream.fetch().await {
492                Ok(unit) => stream.units.push(unit),
493                Err(error) => {
494                    if matches!(error, FillError::Changed) {
495                        self.state
496                            .discard(snapshot, &self.subject, &stream.pointer)
497                            .await;
498                    }
499                    return Err(error);
500                }
501            }
502        }
503        Ok(Some(PreparedStream::Cached(stream)))
504    }

506    async fn cached_nonce(
507        &self,
508        snapshot: Snapshot,
509        pointer: Pointer,
510        range: BytesRange,
511    ) -> Option<CachedStream> {
512        let bytes = self.normalized(range, pointer.metadata.content_length())?;
513        let first = bytes.start / self.chunk_size;
514        let last = (bytes.end - 1) / self.chunk_size;
515        let mut stream = CachedStream::new(self.clone(), snapshot, pointer, bytes, Vec::new());
516        if last - first < self.unit_chunks() {
517            for index in (first..=last).rev() {
518                stream.units.push(stream.cached_chunk(index).await?);
519            }
520            return Some(stream);
521        }
522        // Large Tier 4 range: verify all chunks are present without buffering
523        // the full range in memory, then stream lazily from cache.
524        let first_unit = stream.cached_chunk(first).await?;
525        if !stream.has_cached_chunk(last).await {
526            return None;
527        }
528        for index in (first + 1)..last {
529            if !stream.has_cached_chunk(index).await {
530                return None;
531            }
532        }
533        stream.units.push(first_unit);
534        Some(stream)
535    }

537    async fn store_unit(&self, snapshot: Snapshot, pointer: &Pointer, unit: &Unit) {
538        let prefix = data_prefix(
539            snapshot.generation,
540            &self.subject,
541            pointer.identity,
542            self.chunk_size,
543        );
544        let mut offset = 0;
545        let mut index = unit.start / self.chunk_size;
546        while offset < unit.data.len() {
547            let end = offset
548                .saturating_add(self.chunk_size as usize)
549                .min(unit.data.len());
550            let _ = self
551                .state
552                .cache
553                .write(&format!("{prefix}/{index:x}"), unit.data.slice(offset..end))
554                .await;
555            offset = end;
556            index += 1;
557        }
558    }

560    async fn fetch_unit(
561        &self,
562        snapshot: Snapshot,
563        pointer: Pointer,
564        start: u64,
565        count: usize,
566    ) -> FillResult {
567        let length = pointer.metadata.content_length();
568        let end = start
569            .saturating_add(self.chunk_size.saturating_mul(count as u64))
570            .min(length);
571        let cap = self.source.capability();
572        let version = if cap.read_with_version {
573            pointer.metadata.version().or(self.args.version())
574        } else {
575            self.args.version()
576        };
577        let etag = cap
578            .read_with_if_match
579            .then(|| strong_etag(&pointer.metadata))
580            .flatten();
581        let args = if version != self.args.version() || etag.is_some() {
582            let options = ReadOptions {
583                version: version.map(str::to_owned),
584                if_match: etag.map(str::to_owned),
585                ..Default::default()
586            };
587            let (_, args, _) = options.into();
588            args
589        } else {
590            self.args.clone()
591        };
592        let range = BytesRange::new(start, Some(end - start));
593        let origin = self.source.read(&self.context, &self.path, args)?;
594        let (response, mut stream) = origin.open(range).await.map_err(fill_open_error)?;
595        let verified = verify_response(&pointer, response.metadata(), version, etag)?;
596        let data = read_unit(&mut stream, end - start)
597            .await
598            .map_err(fill_open_error)?;
599        if !verified {
600            let stat_version = if cap.stat_with_version {
601                version
602            } else {
603                self.args.version()
604            };
605            self.verify_stat(&pointer, stat_version).await?;
606        }
607        if data.len() as u64 != end - start {
608            return Err(FillError::Changed);
609        }
610        let unit = Unit { start, data };
611        self.store_unit(snapshot, &pointer, &unit).await;
612        Ok(unit)
613    }
614
615    async fn verify_stat(
616        &self,
617        pointer: &Pointer,
618        version: Option<&str>,
619    ) -> std::result::Result<(), FillError> {
620        let metadata = self.stat(version).await.map_err(|error| {
621            if error.kind() == ErrorKind::NotFound {
622                FillError::Changed
623            } else {
624                FillError::Unverifiable(Arc::new(error))
625            }
626        })?;
627        if metadata.content_length() != pointer.metadata.content_length()
628            || Identity::verifiable(&metadata) != Some(pointer.identity)
629        {
630            return Err(FillError::Changed);
631        }
632        Ok(())
633    }
634}

636fn fill_open_error(error: Error) -> FillError {
637    match error.kind() {
638        ErrorKind::ConditionNotMatch | ErrorKind::NotFound | ErrorKind::RangeNotSatisfied => {
639            FillError::Changed
640        }
641        _ => error.into(),
642    }
643}

645fn verify_response(
646    pointer: &Pointer,
647    response: Option<&Metadata>,
648    version: Option<&str>,
649    etag: Option<&str>,
650) -> std::result::Result<bool, FillError> {
651    let expected = &pointer.metadata;
652    if let Some(response) = response
653        && (response.is_file() && response.content_length() != expected.content_length()
654            || matches!((expected.version(), response.version()), (Some(a), Some(b)) if a != b)
655            || matches!((strong_etag(expected), strong_etag(response)), (Some(a), Some(b)) if a != b)
656            || (pointer.identity.tier == Tier::Modified
657                && response.last_modified().is_some()
658                && response.last_modified() != expected.last_modified()))
659    {
660        return Err(FillError::Changed);
661    }
662    let version = version.is_some() && version == expected.version()
663        || response
664            .and_then(Metadata::version)
665            .is_some_and(|v| Some(v) == expected.version());
666    let etag = etag.is_some()
667        || response
668            .and_then(strong_etag)
669            .is_some_and(|v| Some(v) == strong_etag(expected));
670    Ok(match pointer.identity.tier {
671        Tier::Version => version && (strong_etag(expected).is_none() || etag),
672        Tier::Etag => etag,
673        Tier::Modified => response
674            .filter(|metadata| metadata.is_file())
675            .and_then(Metadata::last_modified)
676            .is_some_and(|v| Some(v) == expected.last_modified()),
677        Tier::Nonce => false,
678    })
679}

681async fn read_unit(stream: &mut Box<dyn ReadStreamDyn>, expected: u64) -> Result<Buffer> {
682    let mut bytes = 0_u64;
683    let mut parts = Vec::new();
684    loop {
685        let next = stream.read().await?;
686        if next.is_empty() {
687            return Ok(parts.into_iter().flatten().collect());
688        }
689        bytes = bytes.saturating_add(next.len() as u64);
690        if bytes > expected {
691            return Err(Error::new(
692                ErrorKind::Unexpected,
693                "origin exceeded the aligned read range",
694            ));
695        }
696        parts.push(next);
697    }
698}

700struct CachedStream {
701    reader: CachedReader,
702    snapshot: Snapshot,
703    pointer: Pointer,
704    prefix: String,
705    remaining: Range<u64>,
706    units: Vec<Unit>,
707    terminal: Option<FillError>,
708}
709
710impl CachedStream {
711    fn new(
712        reader: CachedReader,
713        snapshot: Snapshot,
714        pointer: Pointer,
715        remaining: Range<u64>,
716        units: Vec<Unit>,
717    ) -> Self {
718        let prefix = data_prefix(
719            snapshot.generation,
720            &reader.subject,
721            pointer.identity,
722            reader.chunk_size,
723        );
724        Self {
725            reader,
726            snapshot,
727            pointer,
728            prefix,
729            remaining,
730            units,
731            terminal: None,
732        }
733    }
734
735    async fn cached_chunk(&self, index: u64) -> Option<Unit> {
736        let start = index * self.reader.chunk_size;
737        let expected = (self.pointer.metadata.content_length() - start).min(self.reader.chunk_size);
738        let data = self
739            .reader
740            .state
741            .cache
742            .read(&format!("{}/{index:x}", self.prefix))
743            .await
744            .ok()?;
745        (data.len() as u64 == expected).then_some(Unit { start, data })
746    }
747
748    async fn has_cached_chunk(&self, index: u64) -> bool {
749        let start = index * self.reader.chunk_size;
750        let expected = (self.pointer.metadata.content_length() - start).min(self.reader.chunk_size);
751        let key = format!("{}/{index:x}", self.prefix);
752        if self.reader.state.cache.info().capability().stat {
753            self.reader
754                .state
755                .cache
756                .stat(&key)
757                .await
758                .is_ok_and(|meta| meta.is_file() && meta.content_length() == expected)
759        } else {
760            self.cached_chunk(index).await.is_some()
761        }
762    }
763
764    async fn fetch(&self) -> FillResult {
765        let chunk_size = self.reader.chunk_size;
766        let first = self.remaining.start / chunk_size;
767        if let Some(hit) = self.cached_chunk(first).await {
768            return Ok(hit);
769        }
770        if self.pointer.identity.tier == Tier::Nonce {
771            return Err(FillError::Changed);
772        }
773        let last = ((self.remaining.end - 1) / chunk_size)
774            .min(first.saturating_add(self.reader.unit_chunks() - 1));
775        let mut keys = Vec::new();
776        for index in first..=last {
777            if index != first && self.cached_chunk(index).await.is_some() {
778                break;
779            }
780            keys.push(FillKey {
781                epoch: self.snapshot.epoch,
782                key: format!("{}/{index:x}", self.prefix),
783            });
784        }
785        let count = keys.len();
786        let reader = self.reader.clone();
787        let pointer = self.pointer.clone();
788        let snapshot = self.snapshot;
789        match self
790            .reader
791            .state
792            .fills
793            .run(
794                self.reader.context.executor(),
795                keys,
796                move |count| async move {
797                    reader
798                        .fetch_unit(snapshot, pointer, first * chunk_size, count)
799                        .await
800                },
801            )
802            .await
803        {
804            Some(result) => result,
805            None => {
806                // If the executor dropped the shared fill task without running it
807                // to completion, fall back to fetching the unit inline.
808                self.reader
809                    .fetch_unit(
810                        self.snapshot,
811                        self.pointer.clone(),
812                        first * chunk_size,
813                        count,
814                    )
815                    .await
816            }
817        }
818    }
819
820    async fn next(&mut self) -> std::result::Result<Buffer, FillError> {
821        if let Some(error) = &self.terminal {
822            return Err(error.clone());
823        }
824        if self.remaining.is_empty() {
825            return Ok(Buffer::new());
826        }
827        let unit = if let Some(unit) = self.units.pop() {
828            unit
829        } else {
830            match self.fetch().await {
831                Ok(unit) => unit,
832                Err(error) => {
833                    self.terminal = Some(error.clone());
834                    if matches!(error, FillError::Changed) {
835                        self.reader
836                            .state
837                            .discard(self.snapshot, &self.reader.subject, &self.pointer)
838                            .await;
839                    }
840                    return Err(error);
841                }
842            }
843        };
844        let start = self.remaining.start;
845        let end = self.remaining.end.min(unit.end());
846        self.remaining.start = end;
847        Ok(unit
848            .data
849            .slice((start - unit.start) as usize..(end - unit.start) as usize))
850    }
851}

853struct NonceFillStream {
854    reader: CachedReader,
855    snapshot: Snapshot,
856    pointer: Pointer,
857    publish_pointer: bool,
858    prefix: String,
859    inner: Box<dyn ReadStreamDyn>,
860    aligned_offset: u64,
861    aligned_end: u64,
862    requested: Range<u64>,
863    chunk_index: u64,
864    chunk_parts: Vec<Buffer>,
865    chunk_len: usize,
866    done: bool,
867    terminal: Option<FillError>,
868}
869
870impl NonceFillStream {
871    async fn flush_chunk(&mut self) {
872        if self.chunk_len == 0 {
873            return;
874        }
875        let data: Buffer = self.chunk_parts.drain(..).flatten().collect();
876        self.chunk_len = 0;
877        if self.reader.state.epoch(&self.reader.subject).await == self.snapshot.epoch {
878            let _ = self
879                .reader
880                .state
881                .cache
882                .write(&format!("{}/{:x}", self.prefix, self.chunk_index), data)
883                .await;
884        }
885        self.chunk_index += 1;
886    }
887
888    async fn finish(&mut self) {
889        self.flush_chunk().await;
890        if self.publish_pointer {
891            self.reader
892                .state
893                .publish(self.snapshot, &self.reader.subject, &self.pointer)
894                .await;
895        }
896        self.done = true;
897    }
898
899    async fn next(&mut self) -> std::result::Result<Buffer, FillError> {
900        if let Some(error) = &self.terminal {
901            return Err(error.clone());
902        }
903        if self.done {
904            return Ok(Buffer::new());
905        }
906        let chunk_size = self.reader.chunk_size as usize;
907        loop {
908            let buf = match self.inner.read().await {
909                Ok(buf) => buf,
910                Err(error) => {
911                    let error = fill_open_error(error);
912                    if matches!(error, FillError::Changed) && !self.publish_pointer {
913                        self.reader
914                            .state
915                            .discard(self.snapshot, &self.reader.subject, &self.pointer)
916                            .await;
917                    }
918                    self.terminal = Some(error.clone());
919                    return Err(error);
920                }
921            };
922            if buf.is_empty() {
923                if self.aligned_offset != self.aligned_end {
924                    let error = FillError::Changed;
925                    if !self.publish_pointer {
926                        self.reader
927                            .state
928                            .discard(self.snapshot, &self.reader.subject, &self.pointer)
929                            .await;
930                    }
931                    self.terminal = Some(error.clone());
932                    return Err(error);
933                }
934                self.finish().await;
935                return Ok(Buffer::new());
936            }
937            let buf_start = self.aligned_offset;
938            let buf_end = buf_start.saturating_add(buf.len() as u64);
939            if buf_end > self.aligned_end {
940                let error = FillError::from(Error::new(
941                    ErrorKind::Unexpected,
942                    "origin exceeded the aligned read range",
943                ));
944                self.terminal = Some(error.clone());
945                return Err(error);
946            }
947            self.aligned_offset = buf_end;
948
949            let mut remaining = buf.clone();
950            while !remaining.is_empty() {
951                let need = chunk_size - self.chunk_len;
952                let take = need.min(remaining.len());
953                self.chunk_parts.push(remaining.slice(..take));
954                self.chunk_len += take;
955                remaining = remaining.slice(take..);
956                if self.chunk_len == chunk_size {
957                    self.flush_chunk().await;
958                }
959            }
960
961            if self.aligned_offset == self.aligned_end {
962                self.finish().await;
963            }
964
965            let out_start = buf_start.max(self.requested.start);
966            let out_end = buf_end.min(self.requested.end);
967            if out_start < out_end {
968                return Ok(
969                    buf.slice((out_start - buf_start) as usize..(out_end - buf_start) as usize)
970                );
971            }
972            if self.done {
973                return Ok(Buffer::new());
974            }
975        }
976    }
977}

979enum PreparedStream {
980    Cached(CachedStream),
981    NonceFill(NonceFillStream),
982}
983
984impl PreparedStream {
985    fn pointer(&self) -> &Pointer {
986        match self {
987            Self::Cached(stream) => &stream.pointer,
988            Self::NonceFill(stream) => &stream.pointer,
989        }
990    }
991
992    fn into_pointer(self) -> Pointer {
993        match self {
994            Self::Cached(stream) => stream.pointer,
995            Self::NonceFill(stream) => stream.pointer,
996        }
997    }
998
999    async fn next(&mut self) -> std::result::Result<Buffer, FillError> {
1000        match self {
1001            Self::Cached(stream) => stream.next().await,
1002            Self::NonceFill(stream) => stream.next().await,
1003        }
1004    }
1005}

1007impl ReadStream for PreparedStream {
1008    async fn read(&mut self) -> Result<Buffer> {
1009        self.next().await.map_err(FillError::into_error)
1010    }
1011}
1012
1013impl Read for CachedReader {
1014    async fn open(&self, range: BytesRange) -> Result<(RpRead, Box<dyn ReadStreamDyn>)> {
1015        let mut prepared = self.prepare(range).await;
1016        if matches!(prepared, Err(FillError::Changed)) {
1017            prepared = self.prepare(range).await;
1018        }
1019        match prepared {
1020            Ok(Some(stream)) => Ok((
1021                RpRead::new(stream.pointer().metadata.clone()),
1022                Box::new(stream),
1023            )),
1024            Ok(None) | Err(FillError::Unverifiable(_)) => self.origin()?.open(range).await,
1025            Err(error) => Err(error.into_error()),
1026        }
1027    }
1028
1029    async fn read(&self, range: BytesRange) -> Result<(RpRead, Buffer)> {
1030        let Some(expected) = range.size() else {
1031            return self.origin()?.read(range).await;
1032        };
1033        let mut last_error = FillError::Changed;
1034        for attempt in 0..2 {
1035            let mut stream = match self.prepare(range).await {
1036                Ok(Some(stream)) => stream,
1037                Ok(None) | Err(FillError::Unverifiable(_)) => {
1038                    return self.origin()?.read(range).await;
1039                }
1040                Err(error @ FillError::Changed) if attempt == 0 => {
1041                    last_error = error;
1042                    continue;
1043                }
1044                Err(error) => return Err(error.into_error()),
1045            };
1046            let mut parts = Vec::new();
1047            let mut retry = false;
1048            loop {
1049                match stream.next().await {
1050                    Ok(part) if part.is_empty() => break,
1051                    Ok(part) => parts.push(part),
1052                    Err(error @ FillError::Changed) if attempt == 0 => {
1053                        last_error = error;
1054                        retry = true;
1055                        break;
1056                    }
1057                    Err(FillError::Unverifiable(_)) => return self.origin()?.read(range).await,
1058                    Err(error) => return Err(error.into_error()),
1059                }
1060            }
1061            if retry {
1062                continue;
1063            }
1064            let data: Buffer = parts.into_iter().flatten().collect();
1065            let bound = if range.is_suffix() {
1066                expected.min(stream.pointer().metadata.content_length())
1067            } else {
1068                expected
1069            };
1070            if data.len() as u64 != bound {
1071                // Defer exact-read EOF error classification to the source.
1072                return self.origin()?.read(range).await;
1073            }
1074            return Ok((RpRead::new(stream.into_pointer().metadata), data));
1075        }
1076        Err(last_error.into_error())
1077    }
